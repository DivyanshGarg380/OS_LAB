	int n, m;
	// Enter number of processes -> scanf("%d", &n);
	// Enter number of resources -> scanf("%d", &m);

	int alloc[n][m], max[n][m] , need[n][m], avail[m];
	
	// Enter allocation matrix -> scanf("%d", &alloc[i][j]);
	// Enter max matrix -> scanf("%d", &max[i][j]);
	// Enter available vector -> scanf("%d", &avail[i]);

	// compute need matrix -> need[i][j] = max[i][j] - alloc[i][j];
	
	int finish[n], safeSeq[n], work[m];

	// finish[i] = 0 -> loops till n
	// work[i] = avail[i] -> loops till m

	int cnt = 0;
	while(cnt < n) {
		int found = 0;
		for(int i = 0; i < n; ++i) {
			if(!finish[i]) {
				int flag = 1;
				for(int j = 0; j < m; ++j) {
					if(need[i][j] > work[j]) // set flag = 0 and break
				}

				if(flag) {
					for(int j = 0; j < m; ++j) work[j] += alloc[i][j];
					safeSeq[cnt++] = i;
					finish[i] = 1;
					found = 1;
				}
			}
		}

		if(found == 0) // system not safe -> return 0
	}

	int request[m];
	// Enter Request vector -> scanf("%d", &req[i]);

	// Check if Request <= Need -> if(req[i] <= need[pid][i])
	// Else throw error -> Request exceeds need

	// Check if Request <= Available -> if(req[i] <= available[i]) 
	// Else throw error -> Resource not available

	// Now Pretend allocation
	for(int i = 0; i < m; ++i) {
		avail[i] -= req[i];
		alloc[pid][i] += req[i];
		need[pid][i] -= req[i];
	}

	// Resource granted
	// Check with banker's algo if this allocation is safe.
	// If not then revert back the "allocation"
	
	for(int i = 0; i < m; ++i) {
		avail[i] += req[i];
		alloc[pid][i] -= req[i];
		need[pid][i] += req[i];
	}

	// Enter number of processes -> scanf("%d", &n);
	// Enter number of resources -> scanf("%d", &m);

	// Enter allocation matrix -> scanf("%d", &alloc[i][j]);
	// Enter request matrix -> scanf("%d", &req[i][j]);
	// Enter available vector -> scanf("%d", &avail[i]);

	// compute need matrix -> need[i][j] = max[i][j] - alloc[i][j];
	
	int finish[n];
	// if allocation = 0 -> mark finish = 1;
	for(int i = 0; i < n; ++i) {
		int sum = 0;
		for(int j = 0; j < m; ++j) sum += alloc[i][j];
		if(sum == 0) finish[i] = 1;
		else finish[i] = 0;
	}

	int work[m];
	// Set work[i] = avail[i];

	int found = 0;
	
	do {
		found = 0;
		for(int i = 0; i < n; ++i) {
			if(!finish[i]) {
				int flag = 1;
				for(int j = 0; j < m; ++j) {
					if(req[i][j] > work[j]) // Set flag = 0 and break
				}
	
				if(flag) {
					for(int j = 0; j < m; ++j) work[j] += alloc[i][j];
					finish[i] = 1;
					found = 1;
				}
			}
		}
	} while(found);

	// check for deadlock now
	int deadlock = 0;
	for(int i = 0; i < n; ++i) {
		if(!finish[i]) deadlock = 1; // Process i is in deadlock
	}

	if(!deadlock) // No deadlock
}

void fifo(int pages[], int n, int capacity) {
	int frames[MAX], front = 0, faults = 0;

	// Set frames[i] = -1 -> loop till capacity
	
	for(int i = 0; i < n; ++i) {
		int found = 0;
		for(int j = 0; j < capacity; ++j) {
			if(frames[j] == pages[i] // set found = 1 and break

		}
		
		if(!found) {	
			frames[front] = pages[i];
			front = (front + 1) % capacity;
			faults++;
		}
	}

}

int predict(int pages[], int frames[], int n, int idx, int capacity) {
	int farthest = idx, pos = -1;
	
	for(int i = 0; i < capacity; ++i) {
		int j;
		for(j = idx; j < n; ++j) {
			if(frames[i] == pages[j]) {
				if(j > farthest) // set farthest = j and pos = i;					break;
			}
		}

		if(j == n) return i;
	}
	return (pos == -1) ? 0 : pos;	
}	

void optimal(int pages[], in n, int capacity) {
	int frames[MAX], faults = 0, size = 0;
	// Set frames[i] = -1 for all capacity

	for(int i = 0; i < n; ++i) {
		int found = 0;
		for(int j = 0; j < size; ++j) {
			if(frames[j] == pages[i]) // set found = 1 and break

		}

		if(!found) {
			if(size < capacity) frames[size++] = pages[i];

				int pos = predict(pages, frames, n, i + 1, capacity);
				frames[pos] = pages[i];

		}
	}
	// print faults
}

#define MAX 100

void fcfs(int req[], int n, int head) {
	int total = 0;
	for(int i = 0; i < n; ++i) {
		total += abs(req[i] - head);
		head = req[i];
	}

}

void sstf(int req[], int n, int head) {
	int visited[MAX] = {0};
	int total = 0;

	for(int i = 0; i < n; ++i) {
		int mini = INT_MAX, idx = -1;
		for(int j = 0; j < n; ++j) {
			if(!visited[j]) {
				int dist = abs(req[j] - head);
				if(dist < mini) //  set mini = dist and idx = j;
			}
		}
		total += mini;
		head = req[idx];
		visited[idx] = 1;
	}

	// print total
}

void scan(int req[], int n, int head, int size) {
	int total;
	sort(req, n) // apply bubble sort for simplicity

	int i;
	for(i = 0; i < n; ++i) if(req[i] >= head) break;
	
	// move right
	for(int j = i; j < n; ++j) {
		total += abs(req[j] - head);
		head = req[j];
	}

	// move to end now
	total += abs(head - size + 1);
	head = size - 1;

	// move left now
	for(int j = i - 1; j >= 0; --j) {
		total += abs(req[j] - head);
		head = req[j];
	}

	// print total
}

void cscan(int req[], int n, int head, int size) {
	int total = 0;
	
	sort(req, n);
	int i;
	for(int i = 0; i < n; ++i) if(req[i] >= head) break;

	// move right
	for(int j = i; j < n; ++j) {
		total += abs(req[j] - head);
		head = req[j];
	}

	// move to end
	total += abs(head - size + 1);
	// jump to 0 now
	total += size - 1;
	head = 0;

	// move right again
	for(int j = 0; j < i; ++j) {
		total += abs(head - req[j]);
		head = req[j];
	}

	// print total
}

void look(int req[], int n, int head) {
	int total = 0;
	sort(req, n);
	int i;
	for(i = 0; i < n; ++i) if(req[i] >= head) break;

	// move right
	for(int j = i; j < n; ++j) {
		total += abs(head - req[j]);
		head = req[j];
	}

	// reverse now
	for(int j = i - 1; j >= 0; --j) {
		total += abs(req[j] - head);
		head = req[j];
	}

	// print total
}

void clook(int req[], int n, int head) {
	int total = 0;
	sort(req, n);
	int i;
	for(i = 0; i < n; ++i) if(req[i] >= head) break;

	// move right
	for(int j = i; j < n; ++j) {	
		total += abs(req[j] - head);
		head = req[j];
	}

	// jump to smallest req
	if(i > 0) {
		total += abs(head-req[0]);
		head = req[0];
	}
	
	// continue same direction
	for(int j = 1; j < i; ++j) {
		total += abs(head - req[j]);
		head = req[j];
	}
